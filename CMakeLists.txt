cmake_minimum_required(VERSION 3.22)
project(YaST VERSION 24.0.144 LANGUAGES C)

if (NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
endif ()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options($<$<C_COMPILER_ID:MSVC>:/utf-8>)
option(YAST_MOCKS_MODE "Run with synthetic data instead of accessing FM24" OFF)
option(YAST_USE_LOCAL_CONFIG "Read the developer's optional src/app/config.h" ON)
option(YAST_UI_TESTS "Run native graphics/input checks (requires a desktop session)" OFF)
include(cmake/dependencies.cmake)
find_package(Threads REQUIRED)

if (WIN32)
	set(PLATFORM_SOURCE src/platform/windows.c)
	set(PLATFORM_DEFINE ARCH_WIN)
elseif (APPLE)
	set(PLATFORM_SOURCE src/platform/macos.c)
	set(PLATFORM_DEFINE ARCH_MACOS)
elseif (CMAKE_SYSTEM_NAME STREQUAL "Linux")
	set(PLATFORM_SOURCE src/platform/linux.c)
	set(PLATFORM_DEFINE ARCH_LINUX)
else ()
	message(FATAL_ERROR "Unsupported platform: ${CMAKE_SYSTEM_NAME}")
endif ()

add_library(yast_core STATIC
	${PLATFORM_SOURCE} src/platform/thread.c src/platform/paths.c
	src/core/logger.c src/app/cache.c src/app/options.c src/app/injury-names.c
	src/app/player.c src/app/player-snapshot.c src/app/player-table.c
	src/app/search.c src/app/search-handler.c src/app/analysis.c src/app/autocomplete.c
	src/app/helpers/date.c src/app/helpers/file-watcher.c src/app/helpers/formatter.c
	src/app/helpers/game.c src/app/helpers/ratings.c src/app/helpers/text.c
	src/app/helpers/vector-shared-pointer.c
)
target_include_directories(yast_core PUBLIC src)
target_link_libraries(yast_core PUBLIC Threads::Threads utf8proc)
target_compile_definitions(yast_core PUBLIC ${PLATFORM_DEFINE} _GNU_SOURCE
	APP_NAME="${PROJECT_NAME}" APP_VERSION="${PROJECT_VERSION}"
	$<$<CONFIG:Debug>:DEBUG>
	$<$<BOOL:${YAST_MOCKS_MODE}>:MOCKS_MODE>
)
if (YAST_USE_LOCAL_CONFIG AND EXISTS "${PROJECT_SOURCE_DIR}/src/app/config.h")
	target_compile_definitions(yast_core PUBLIC YAST_USE_LOCAL_CONFIG)
endif ()
if (WIN32)
	target_link_libraries(yast_core PUBLIC shell32 ole32)
elseif (UNIX)
	target_link_libraries(yast_core PUBLIC m)
endif ()

file(GLOB_RECURSE RESOURCE_INPUTS CONFIGURE_DEPENDS RELATIVE "${PROJECT_SOURCE_DIR}"
	assets/flags/*.png assets/icons/*.png assets/fonts/*.ttf assets/fonts/*.otf)
list(APPEND RESOURCE_INPUTS assets/branding/png/128.png)
set(RESOURCE_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/resources.c")
add_custom_command(OUTPUT "${RESOURCE_OUTPUT}"
	COMMAND "${CMAKE_COMMAND}" "-DROOT=${PROJECT_SOURCE_DIR}" "-DOUTPUT=${RESOURCE_OUTPUT}"
	"-DINPUTS=${RESOURCE_INPUTS}" -P "${PROJECT_SOURCE_DIR}/cmake/resources.cmake"
	DEPENDS ${RESOURCE_INPUTS} cmake/resources.cmake VERBATIM)
add_library(yast_ui STATIC src/app/ui.c src/app/ui/controls.c src/app/ui/search.c
	src/app/ui/player-info.c src/app/ui/best-eleven.c src/app/ui/squad-depth.c
	src/app/helpers/icons.c "${RESOURCE_OUTPUT}")
target_include_directories(yast_ui SYSTEM PUBLIC "${raygui_SOURCE_DIR}/src")
target_link_libraries(yast_ui PUBLIC yast_core raylib)
add_executable(yast src/main.c)
target_link_libraries(yast PRIVATE yast_ui)
set_target_properties(yast PROPERTIES OUTPUT_NAME "${PROJECT_NAME}")
foreach(target IN ITEMS yast yast_ui yast_core)
	target_compile_options(${target} PRIVATE
		$<$<C_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra;-Wpedantic;-Wshadow;-Wconversion;-Wstrict-prototypes;-Wmissing-prototypes;-Wno-missing-field-initializers>
		$<$<C_COMPILER_ID:MSVC>:/W4>)
endforeach()
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo LANGUAGES C)
if (ipo)
	set_property(TARGET yast yast_ui yast_core PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif ()
if (APPLE)
	set_target_properties(yast PROPERTIES MACOSX_BUNDLE TRUE
		MACOSX_BUNDLE_ICON_FILE yast MACOSX_BUNDLE_GUI_IDENTIFIER com.philarmstead.yast
		MACOSX_BUNDLE_BUNDLE_NAME "${PROJECT_NAME}"
		MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
		MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
	target_sources(yast PRIVATE assets/branding/yast.icns)
	set_source_files_properties(assets/branding/yast.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
	target_link_options(yast PRIVATE $<$<CONFIG:Release>:-Wl,-x>)
elseif (WIN32)
	enable_language(RC)
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/app-icon.rc" "1 ICON \"${PROJECT_SOURCE_DIR}/assets/branding/yast.ico\"\n")
	target_sources(yast PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app-icon.rc")
	target_link_options(yast PRIVATE $<$<AND:$<CONFIG:Release>,$<C_COMPILER_ID:GNU,Clang>>:-mwindows>)
	if (MSVC)
		set_target_properties(yast PROPERTIES WIN32_EXECUTABLE TRUE)
		target_link_options(yast PRIVATE /ENTRY:mainCRTStartup)
	endif ()
endif ()
include(GNUInstallDirs)
if (APPLE)
	install(TARGETS yast BUNDLE DESTINATION .)
else ()
	install(TARGETS yast RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif ()
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
	install(FILES assets/branding/com.philarmstead.yast.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
	install(FILES assets/branding/yast.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps" RENAME com.philarmstead.yast.svg)
endif ()
set(license_destination "${CMAKE_INSTALL_DATADIR}/yast/licenses")
if (APPLE)
	set(license_destination "YaST.app/Contents/Resources/licenses")
endif ()
function(yast_license source name)
	configure_file("${source}" "${CMAKE_CURRENT_BINARY_DIR}/licenses/${name}" COPYONLY)
	install(FILES "${CMAKE_CURRENT_BINARY_DIR}/licenses/${name}" DESTINATION "${license_destination}")
endfunction()
yast_license("${PROJECT_SOURCE_DIR}/COPYING" COPYING)
yast_license("${PROJECT_SOURCE_DIR}/docs/CREDITS.md" CREDITS.md)
yast_license("${PROJECT_SOURCE_DIR}/assets/fonts/OFL.txt" NotoSans-OFL.txt)
yast_license("${PROJECT_SOURCE_DIR}/assets/fonts/Unifont-LICENSE.txt" Unifont-LICENSE.txt)
yast_license("${raylib_SOURCE_DIR}/LICENSE" raylib-LICENSE)
yast_license("${raygui_SOURCE_DIR}/LICENSE" raygui-LICENSE)
yast_license("${utf8proc_SOURCE_DIR}/LICENSE.md" utf8proc-LICENSE.md)
yast_license("${raylib_SOURCE_DIR}/src/external/glfw/LICENSE.md" GLFW-LICENSE.md)
yast_license("${PROJECT_SOURCE_DIR}/assets/licenses/raylib-third-party.txt" raylib-third-party.txt)
if (APPLE)
	add_custom_command(TARGET yast POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_directory
		"${CMAKE_CURRENT_BINARY_DIR}/licenses" "$<TARGET_BUNDLE_DIR:yast>/Contents/Resources/licenses")
endif ()
add_custom_target(run COMMAND $<TARGET_FILE:yast> DEPENDS yast WORKING_DIRECTORY $<TARGET_FILE_DIR:yast> USES_TERMINAL)
add_custom_target(info COMMAND "${CMAKE_COMMAND}" -E echo "Compiler: ${CMAKE_C_COMPILER_ID} / C17, target: $<TARGET_FILE:yast>")
if (WIN32)
	include(cmake/windows.cmake)
endif ()
include(CTest)
if (BUILD_TESTING)
	add_executable(yast_tests tests/core.c)
	target_link_libraries(yast_tests PRIVATE yast_core)
	target_compile_definitions(yast_tests PRIVATE TEST_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}")
	target_compile_options(yast_tests PRIVATE $<$<C_COMPILER_ID:GNU,Clang,AppleClang>:-UNDEBUG> $<$<C_COMPILER_ID:MSVC>:/UNDEBUG>)
	add_test(NAME core COMMAND yast_tests)
	if (UNIX)
		find_package(Python3 COMPONENTS Interpreter QUIET)
		if (Python3_Interpreter_FOUND)
			add_test(NAME options COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/options.py" "$<TARGET_FILE:yast_tests>")
		endif ()
	endif ()
	if (YAST_UI_TESTS)
		add_executable(yast_ui_tests tests/ui.c)
		target_link_libraries(yast_ui_tests PRIVATE yast_ui)
		target_compile_options(yast_ui_tests PRIVATE $<$<C_COMPILER_ID:GNU,Clang,AppleClang>:-UNDEBUG> $<$<C_COMPILER_ID:MSVC>:/UNDEBUG>)
		add_test(NAME ui COMMAND yast_ui_tests)
	endif ()
endif ()
